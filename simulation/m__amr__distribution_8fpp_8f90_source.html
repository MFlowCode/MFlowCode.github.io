591 !> Cost-weighted SFC partition of n items into num_procs contiguous Morton-key ranges: walk the key-sorted items (stable merge
592 !! sort) and advance the owner rank when the running weight crosses the next even share. Writes owner(1:n) and
593 !! cut(0:num_procs-1) (running Morton upper bound per rank; ranks with no item inherit the predecessor's bound, so cut is
594 !! non-decreasing).
595 subroutine s_amr_sfc_cut(keys, wt, n, cut, owner)
596
597 integer, intent(in) :: n
598 integer(kind=8), intent(in) :: keys(n)
599 real(wp), intent(in) :: wt(n)
600 integer(kind=8), intent(out) :: cut(0:num_procs - 1)
601 integer, intent(out) :: owner(n)
602 integer :: ord(n), k, r
603 real(wp) :: total, cum, tgt, tol
604
605 cut = -1_8
606 if (n < 1) return
607
608 call s_amr_sort_by_key(keys, n, ord)
609
610 total = 0._wp
611 do k = 1, n
612 total = total + wt(k)
613 end do
614
615 ! chains-on-chains in SFC order; replicated weights in a fixed order, so every rank computes the identical assignment
616 r = 0; cum = 0._wp
617 do k = 1, n
618 tgt = real(r + 1, wp)*total/real(num_procs, wp)
619 ! tolerance = accumulated rounding bound: at an exact share boundary cum and tgt differ by rounding, and a fractional
620 ! cost term could otherwise split identical weights unevenly (e.g. 5/3 instead of 4/4)
621 tol = spacing(tgt)*real(n, wp)
622 if (cum >= tgt - tol .and. r < num_procs - 1) r = r + 1
623 owner(ord(k)) = r
624 cut(r) = keys(ord(k)) ! items visited in ascending Morton key => running upper bound for rank r
625 cum = cum + wt(ord(k))
626 end do
627 ! ranks with no item inherit the predecessor's bound so the search never lands on them
628 do r = 1, num_procs - 1
629 if (cut(r) < cut(r - 1)) cut(r) = cut(r - 1)
630 end do

632 end subroutine s_amr_sfc_cut

634 !> ord(1:n) = item indices in ascending key order. Stable iterative bottom-up merge sort, a pure function of the input so every
635 !! rank derives identical orders from replicated keys (differing assignments would deadlock or abort).

668 end do
